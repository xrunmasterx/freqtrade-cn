def _load(name: str):
    return contract.load_canonical_json_bytes((HERE / name).read_bytes())


def _canonical(value: object) -> bytes:
    return contract.canonical_json_bytes(value)


def _walk_strings(value: object):
    if isinstance(value, dict):
        for key, nested in value.items():
            yield key
            yield from _walk_strings(nested)
    elif isinstance(value, list):
        for nested in value:
            yield from _walk_strings(nested)
    elif isinstance(value, str):
        yield value


def _shannon_entropy(value: str) -> float:
    if not value:
        return 0.0
    return -sum(
        (value.count(character) / len(value))
        * math.log2(value.count(character) / len(value))
        for character in set(value)
    )


class CanonicalJsonTests(unittest.TestCase):
    def test_owned_inventory_is_exact(self):
        self.assertEqual(
            {path.name for path in HERE.iterdir()},
            {
                "contract.py",
                "test_contract.py",
                "effective-config.json",
                "b-request.json",
                "c-request-template.json",
            },
        )

    def test_json_files_are_canonical_utf8_with_one_lf(self):
        for name in EXPECTED_JSON_HASHES:
            with self.subTest(name=name):
                raw = (HERE / name).read_bytes()
                self.assertFalse(raw.startswith(b"\xef\xbb\xbf"))
                self.assertTrue(raw.endswith(b"\n"))
                self.assertFalse(raw.endswith(b"\n\n"))
                self.assertNotIn(b"\r", raw)
                self.assertEqual(raw, _canonical(contract.load_canonical_json_bytes(raw)))

    def test_noncanonical_and_ambiguous_json_is_rejected(self):
        raw = (HERE / "b-request.json").read_bytes()
        value = json.loads(raw)
        variants = [
            raw[:-1],
            raw + b"\n",
            b"\xef\xbb\xbf" + raw,
            raw[:-1] + b"\r\n",
            json.dumps(value, indent=2, sort_keys=True).encode("utf-8") + b"\n",
            raw.replace(b":", b": ", 1),
            b'{"strategy":"VolatilitySystem","strategy":"VolatilitySystem"}\n',
            b'{"value":NaN}\n',
        ]
        for candidate in variants:
            with self.subTest(candidate=candidate[:40]):
                with self.assertRaises(contract.ContractError):
                    contract.load_canonical_json_bytes(candidate)

    def test_sha256_and_size_helpers_are_deterministic(self):
        first = {
            path.name: contract.file_sha256_size(path)
            for path in sorted(HERE.iterdir())
            if path.is_file()
        }
        second = {
            path.name: contract.file_sha256_size(path)
            for path in sorted(HERE.iterdir())
            if path.is_file()
        }
        self.assertEqual(first, second)
        for name, expected_sha256 in EXPECTED_JSON_HASHES.items():
            with self.subTest(name=name):
                sha256, size = first[name]
                self.assertEqual(sha256, expected_sha256)
                self.assertEqual((sha256, size), contract.sha256_size((HERE / name).read_bytes()))

    def test_requests_have_no_credentials_or_high_entropy_placeholders(self):
        values = [_load(name) for name in EXPECTED_JSON_HASHES]
        for value in values:
            serialized = json.dumps(value, sort_keys=True)
            self.assertNotRegex(serialized, r"[0-9a-fA-F]{32,}")
            for item in _walk_strings(value):
                if item in {"__SET_VIA_SECRET_FILE__", "__B_RESULT_STEM__"}:
                    continue
                self.assertNotIn("PLACEHOLDER", item.upper())
                self.assertFalse(len(item) >= 48 and _shannon_entropy(item) >= 4.8, item)
