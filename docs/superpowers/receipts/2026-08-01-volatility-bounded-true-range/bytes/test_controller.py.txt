def canonical(value: object) -> bytes:
    return (
        json.dumps(
            value,
            allow_nan=False,
            ensure_ascii=False,
            separators=(",", ":"),
            sort_keys=True,
        ).encode("utf-8")
        + b"\n"
    )


def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


class FakeContract:
    def __init__(self) -> None:
        self.calls: list[str] = []

    def load_canonical_json_bytes(self, raw: bytes) -> object:
        self.calls.append("load")
        value = json.loads(raw)
        if canonical(value) != raw:
            raise ValueError("not canonical")
        return value

    def validate_effective_config(self, value: object) -> None:
        self.calls.append("config")
        if value != {"config": "frozen"}:
            raise ValueError("bad config")

    def validate_b_request(self, value: object) -> None:
        self.calls.append("b")
        if value != {"retain_data_snapshot": True, "role": "B"}:
            raise ValueError("bad B")

    def validate_c_request_template(self, value: object) -> None:
        self.calls.append("c-template")
        if value != {
            "replay_data_snapshot": {"filename": "__B_RESULT_STEM__"},
            "role": "C",
        }:
            raise ValueError("bad C template")

    def validate_bound_c_request(self, value: object) -> None:
        self.calls.append("c-bound")
        if value != {
            "replay_data_snapshot": {"filename": VALID_STEM},
            "role": "C",
        }:
            raise ValueError("bad bound C")
        if "retain_data_snapshot" in value:
            raise ValueError("retain must be absent")

    def validate_result_stem(self, value: object) -> str:
        self.calls.append("stem")
        c_stem = VALID_STEM[:-32] + "f" * 32
        if value not in (VALID_STEM, c_stem):
            raise ValueError("bad stem")
        return str(value)

    def bind_c_request(self, template: bytes, stem: str) -> bytes:
        self.calls.append("bind")
        self.validate_result_stem(stem)
        value = self.load_canonical_json_bytes(template)
        self.validate_c_request_template(value)
        value = copy.deepcopy(value)
        value["replay_data_snapshot"]["filename"] = stem
        self.validate_bound_c_request(value)
        return canonical(value)


class FakeRunner:
    def __init__(self, fixture: "Fixture") -> None:
        self.fixture = fixture
        self.calls: list[tuple[str, ...]] = []
        self.cleanup_calls: list[tuple[str, ...]] = []
        self.fail: dict[str, int] = {}
        self.stdout: dict[str, bytes] = {}
        self.stderr: dict[str, bytes] = {}
        self.inspect_mutation = None
        self.cleanup_failure = False
        self.cleanup_exception = False
        self.publish_before_ready = False
        self.report_mutation = None

    def run(self, argv: tuple[str, ...]) -> controller.CommandResult:
        self.calls.append(argv)
        label = self._label(argv)
        if label == "network-create":
            return self._result(label, b"network-id\n")
        if label == "network-inspect":
            return self._result(
                label,
                canonical(
                    [
                        {
                            "Driver": "bridge",
                            "EnableIPv6": False,
                            "Internal": False,
                            "Name": self.fixture.config.network_name,
                        }
                    ]
                ),
            )
        if label.startswith("server-"):
            return self._result(label, b"container-id\n")
        if label.startswith("inspect-"):
            role = label.rsplit("-", 1)[1]
            value = self.fixture.inspect_value(role)
            if self.inspect_mutation is not None:
                self.inspect_mutation(value, role)
            return self._result(label, canonical([value]))
        if label.startswith("ready-"):
            if self.publish_before_ready:
                self.fixture.publish_role(label.rsplit("-", 1)[1])
            return self._result(label)
        if label.startswith("post-"):
            role = label.rsplit("-", 1)[1]
            if self.fail.get(label, 0) == 0:
                self.fixture.publish_role(role)
            return self._result(label)
        if label == "validator":
            if self.fail.get(label, 0) == 0:
                self.fixture.publish_report(
                    self.stdout.get(label, b"METRIC-SURVIVOR_PENDING_ANALYSES\n")
                )
            return self._result(
                label,
                self.stdout.get(label, b"METRIC-SURVIVOR_PENDING_ANALYSES\n"),
                self.stderr.get(label, b""),
            )
        if label.startswith("cleanup-"):
            self.cleanup_calls.append(argv)
            if self.cleanup_exception:
                raise OSError("cleanup unavailable")
            if self.cleanup_failure:
                return controller.CommandResult(1, b"", b"cleanup failed")
            return controller.CommandResult(0, b"", b"")
        raise AssertionError(f"unexpected command: {argv}")

    def _result(self, label: str, stdout: bytes = b"", stderr: bytes = b""):
        return controller.CommandResult(
            self.fail.get(label, 0),
            self.stdout.get(label, stdout),
            self.stderr.get(label, stderr),
        )

    @staticmethod
    def _label(argv: tuple[str, ...]) -> str:
        if argv[1:4] == ("network", "create", "--driver"):
            return "network-create"
        if argv[1:3] == ("network", "inspect"):
            return "network-inspect"
        if argv[1] == "run" and "--network" in argv:
            if "none" in argv:
                return "validator"
            name = argv[argv.index("--name") + 1]
            return f"server-{name.rsplit('-', 1)[-1].upper()}"
        if argv[1] == "inspect":
            return f"inspect-{argv[-1].rsplit('-', 1)[-1].upper()}"
        if argv[1] == "exec":
            role = argv[2].rsplit("-", 1)[-1].upper()
            return f"ready-{role}" if "container-ready" in argv else f"post-{role}"
        if argv[1:3] in (("rm", "-f"), ("network", "rm")):
            return "cleanup-" + argv[1]
        raise AssertionError(f"unclassified command: {argv}")

class HostControllerTests(unittest.TestCase):
    def setUp(self) -> None:
        self.fx = Fixture()

    def tearDown(self) -> None:
        self.fx.close()

    def test_preflight_stages_exact_b4_c1_without_walking_or_mounting_source_root(self):
        self.fx.host.preflight()
        b_files = sorted(
            path.relative_to(self.fx.host.role_paths["B"].data).as_posix()
            for path in self.fx.host.role_paths["B"].data.rglob("*")
            if path.is_file()
        )
        c_files = sorted(
            path.relative_to(self.fx.host.role_paths["C"].data).as_posix()
            for path in self.fx.host.role_paths["C"].data.rglob("*")
            if path.is_file()
        )
        self.assertEqual(
            [
                "futures/BTC_USDT_USDT-1h-funding_rate.feather",
                "futures/BTC_USDT_USDT-1h-futures.feather",
                "futures/BTC_USDT_USDT-1h-mark.feather",
                "futures/leverage_tiers_USDT.json",
            ],
            b_files,
        )
        self.assertEqual(["futures/leverage_tiers_USDT.json"], c_files)
        self.assertNotEqual(self.fx.source_data, self.fx.host.role_paths["B"].data)
        self.assertNotEqual(self.fx.source_data, self.fx.host.role_paths["C"].data)
        for name, expected in self.fx.data.items():
            destination = self.fx.host.role_paths["B"].data / str(expected["relative_path"])
            self.assertEqual(expected["raw_sha256"], sha256(destination.read_bytes()), name)

    def test_restart_rehashes_bind_real_eight_key_identity(self):
        self.fx.host._inputs = {
            item.name: item for item in self.fx.config.inputs
        }
        admission_raw = self.fx.files["admission"].read_bytes()
        restart = json.loads(self.fx.files["restart"].read_bytes())
        self.assertEqual(
            self.fx.data,
            self.fx.host._verify_evidence(admission_raw, canonical(restart)),
        )

        for mutation in ("missing", "extra"):
            with self.subTest(mutation=mutation):
                value = copy.deepcopy(restart)
                if mutation == "missing":
                    value["end_rehashes"].pop("strategy")
                else:
                    value["end_rehashes"]["unexpected"] = "0" * 64
                with self.assertRaisesRegex(controller.ControllerError, "rehash set"):
                    self.fx.host._verify_evidence(admission_raw, canonical(value))

        for key in (
            "baseline_a",
            "baseline_b",
            "raw_funding",
            "raw_futures",
            "raw_leverage",
            "raw_mark",
            "strategy",
        ):
            with self.subTest(mismatch=key):
                value = copy.deepcopy(restart)
                value["end_rehashes"][key] = "0" * 64
                with self.assertRaisesRegex(controller.ControllerError, "rehash mismatch"):
                    self.fx.host._verify_evidence(admission_raw, canonical(value))

        for key in ("baseline_sha256", "candidate_sha256"):
            with self.subTest(global_identity_mismatch=key):
                value = copy.deepcopy(restart)
                value["global_identity"][key] = "0" * 64
                with self.assertRaisesRegex(controller.ControllerError, "source identity"):
                    self.fx.host._verify_evidence(admission_raw, canonical(value))

        malformed = copy.deepcopy(restart)
        malformed["end_rehashes"]["test"] = "not-a-sha256"
        with self.assertRaisesRegex(controller.ControllerError, "invalid SHA-256"):
            self.fx.host._verify_evidence(admission_raw, canonical(malformed))

        inconsistent = copy.deepcopy(restart)
        inconsistent["end_rehashes"]["test"] = "0" * 64
        with self.assertRaisesRegex(controller.ControllerError, "rehash mismatch"):
            self.fx.host._verify_evidence(admission_raw, canonical(inconsistent))

        malformed_global = copy.deepcopy(restart)
        malformed_global["global_identity"]["test_sha256"] = "not-a-sha256"
        with self.assertRaisesRegex(controller.ControllerError, "invalid SHA-256"):
            self.fx.host._verify_evidence(admission_raw, canonical(malformed_global))

    def test_restart_rehash_failure_is_pre_attempt_and_pre_lease(self):
        restart = json.loads(self.fx.files["restart"].read_bytes())
        restart["end_rehashes"].pop("strategy")
        restart_raw = canonical(restart)
        self.fx.files["restart"].write_bytes(restart_raw)

        receipt = json.loads(self.fx.files["receipt"].read_bytes())
        receipt["files"]["restart"] = {
            "bytes": len(restart_raw),
            "sha256": sha256(restart_raw),
        }
        receipt_raw = canonical(receipt)
        self.fx.files["receipt"].write_bytes(receipt_raw)
        bindings = tuple(
            controller.InputFile(
                item.name,
                item.path,
                len(item.path.read_bytes()),
                sha256(item.path.read_bytes()),
            )
            for item in self.fx.config.inputs
        )
        changed = replace(
            self.fx.config,
            receipt=controller.InputFile(
                "receipt",
                self.fx.files["receipt"],
                len(receipt_raw),
                sha256(receipt_raw),
            ),
            expected_receipt_sha256=sha256(receipt_raw),
            inputs=bindings,
        )
        host = controller.HostController(
            changed,
            runner=self.fx.runner,
            contract=self.fx.contract,
        )
        with self.assertRaisesRegex(controller.ControllerError, "rehash set"):
            host.preflight()
        self.assertFalse(host.attempt_root.exists())
        self.assertFalse(self.fx.spent_lease.exists())
        self.assertEqual([], self.fx.runner.calls)

    def test_preflight_is_create_once_and_second_run_refuses_spent_experiment(self):
        self.fx.host.preflight()
        with self.assertRaises(controller.ControllerError):
            self.fx.host.preflight()

        other = replace(
            self.fx.config,
            attempt_id="attempt-002",
            network_name="bounded-tr-attempt-002",
        )
        with self.assertRaisesRegex(controller.ConsumedExperiment, "spent"):
            controller.HostController(
                other, runner=self.fx.runner, contract=self.fx.contract
            ).preflight()
