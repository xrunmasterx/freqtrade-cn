def _canon(value: object) -> bytes:
    return json.dumps(
        value,
        allow_nan=False,
        ensure_ascii=False,
        separators=(",", ":"),
        sort_keys=True,
    ).encode("utf-8") + b"\n"


def _backend_canon(value: object) -> bytes:
    return _canon(value)[:-1]

def _primary_rows(event_count: int = 28) -> tuple[list[OhlcvRow], list[str]]:
    window = WINDOWS["development"]
    first_full = window.prefix_start_ms + HOUR
    last_full = window.end_ms - THREE_HOURS
    slots = list(range(first_full, last_full + 1, THREE_HOURS))
    scored_slots = [slot for slot in slots if slot >= window.start_ms]
    event_slots = [scored_slots[20 + index * 18] for index in range(event_count)]
    sides = ["long", "short", "short"] + [
        "long" if index % 2 else "short" for index in range(3, event_count)
    ]
    deltas: dict[int, float] = {window.start_ms - THREE_HOURS: 5.0}
    for slot, side in zip(event_slots, sides, strict=True):
        deltas[slot] = 5.0 if side == "long" else -5.0
    deltas[event_slots[3] + THREE_HOURS] = 4.0 if sides[3] == "long" else -4.0
    deltas[last_full] = 5.0

    rows = [OhlcvRow(window.prefix_start_ms, 100.0, 100.25, 99.75, 100.0, 1.0)]
    prior_close = 100.0
    for slot in slots:
        target = prior_close + deltas.get(slot, 0.0)
        delta = (target - prior_close) / 3
        value = prior_close
        for offset in range(3):
            next_value = value + delta
            rows.append(
                OhlcvRow(
                    slot + offset * HOUR,
                    value,
                    max(value, next_value) + 0.25,
                    min(value, next_value) - 0.25,
                    next_value,
                    1.0,
                )
            )
            value = next_value
        prior_close = target
    rows.append(
        OhlcvRow(
            window.end_ms,
            prior_close,
            prior_close + 0.25,
            prior_close - 0.25,
            prior_close,
            1.0,
        )
    )
    return rows, sides


def _series(venue: str, candle_type: str, rows: list[OhlcvRow]):
    raw = b"".join(
        ROW.pack(row.timestamp_ms, row.open, row.high, row.low, row.close, row.volume)
        for row in rows
    )
    descriptor = {
        "venue": venue,
        "pair": PAIR,
        "timeframe": "1h",
        "candle_type": candle_type,
        "columns": ["timestamp_ms", "open", "high", "low", "close", "volume"],
        "row_count": len(rows),
        "start_ms": rows[0].timestamp_ms if rows else None,
        "end_ms": rows[-1].timestamp_ms if rows else None,
    }
    descriptor_bytes = _backend_canon(descriptor)
    digest = hashlib.sha256(
        b"freqtrade.logical-ohlcv-series.v1\0"
        + len(descriptor_bytes).to_bytes(8, "big")
        + descriptor_bytes
        + raw
    ).hexdigest()
    return {**descriptor, "series_digest": f"data-series-{digest}"}, raw


def _replay(primary_rows: list[OhlcvRow], *, funding_rate: float = 0.0):
    window = WINDOWS["development"]
    funding_times = list(range(window.prefix_start_ms, window.end_ms + HOUR, HOUR))
    mark_rows = [OhlcvRow(ts, 100.0, 100.0, 100.0, 100.0, 0.0) for ts in funding_times]
    funding_rows = [
        OhlcvRow(ts, funding_rate, funding_rate, funding_rate, funding_rate, 0.0)
        for ts in funding_times
    ]
    material = [
        _series("okx", "futures", primary_rows),
        _series("okx", "funding_rate", funding_rows),
        _series("okx", "mark", mark_rows),
    ]
    series = sorted(
        (item[0] for item in material),
        key=lambda item: (
            item["venue"],
            item["pair"],
            item["timeframe"],
            item["candle_type"],
            item["start_ms"],
            item["end_ms"],
            item["series_digest"],
        ),
    )
    raw_by_digest = {item[0]["series_digest"]: item[1] for item in material}
    snapshot_payload = {
        "schema_version": 1,
        "canonicalization": "freqtrade-logical-ohlcv-v1",
        "hash_algorithm": "sha256",
        "series_count": len(series),
        "row_count": sum(item["row_count"] for item in series),
        "series": series,
    }
    snapshot = {
        "snapshot_id": "data-snapshot-"
        + hashlib.sha256(
            b"freqtrade.data-snapshot-manifest.v1\0" + _backend_canon(snapshot_payload)
        ).hexdigest(),
        **snapshot_payload,
    }
    digest_by_type = {item["candle_type"]: item["series_digest"] for item in series}
    routes = [
        {
            "ordinal": ordinal,
            "source_role": role,
            "pair": PAIR,
            "timeframe": "1h",
            "candle_type": candle_type,
            "series_digest": digest_by_type[candle_type],
        }
        for ordinal, (role, candle_type) in enumerate(
            (("primary", "futures"), ("funding", "funding_rate"), ("mark", "mark"))
        )
    ]
    replay_payload = {
        "schema_version": 1,
        "row_encoding": "freqtrade-logical-ohlcv-rows-v1",
        "data_snapshot": snapshot,
        "routes": routes,
    }
    replay_id = "replay-payload-" + hashlib.sha256(
        b"freqtrade.data-snapshot-replay-manifest.v1\0" + _backend_canon(replay_payload)
    ).hexdigest()
    manifest = {**replay_payload, "replay_payload_id": replay_id}
    binding = {
        "schema_version": 1,
        "snapshot_id": snapshot["snapshot_id"],
        "row_encoding": replay_payload["row_encoding"],
        "replay_payload_id": replay_id,
    }
    members = {
        f"data_snapshot/series/{digest.removeprefix('data-series-')}.rows": raw
        for digest, raw in raw_by_digest.items()
    }
    return manifest, binding, members


def _order(
    *,
    amount: float,
    price: float,
    timestamp: int | None,
    is_entry: bool,
    side: str,
    tag: str | None,
) -> dict[str, object]:
    return {
        "amount": amount,
        "safe_price": price,
        "ft_order_side": side,
        "order_filled_timestamp": timestamp,
        "ft_is_entry": is_entry,
        "ft_order_tag": tag,
        "cost": amount * price,
    }


def _trade(
    ordinal: int,
    side: str,
    open_timestamp: int,
    close_timestamp: int,
    requested_profit: float = 1.0,
    exit_reason: str = "exit_signal",
    *,
    adjustment_timestamp: int | None = None,
    funding_fees: float = 0.0,
    entry_amount: float = 1.0,
) -> dict[str, object]:
    is_short = side == "short"
    entry_side = "sell" if is_short else "buy"
    exit_side = "buy" if is_short else "sell"
    entries = [
        _order(
            amount=entry_amount,
            price=100.0,
            timestamp=open_timestamp,
            is_entry=True,
            side=entry_side,
            tag="bounded-tr",
        )
    ]
    if adjustment_timestamp is not None:
        entries.append(
            _order(
                amount=0.5,
                price=100.0,
                timestamp=adjustment_timestamp,
                is_entry=True,
                side=entry_side,
                tag="adjustment",
            )
        )
    total_amount = sum(float(item["amount"]) for item in entries)
    entry_cash = total_amount * 100.0
    entry_fees = entry_cash * 0.0005
    if is_short:
        exit_price = (entry_cash - entry_fees + funding_fees - requested_profit) / (
            total_amount * 1.0005
        )
    else:
        exit_price = (requested_profit + entry_cash + entry_fees - funding_fees) / (
            total_amount * 0.9995
        )
    exit_order = _order(
        amount=total_amount,
        price=exit_price,
        timestamp=close_timestamp,
        is_entry=False,
        side=exit_side,
        tag=exit_reason,
    )
    direction = -1.0 if is_short else 1.0
    cashflows = [
        direction * (-1.0 if item["ft_is_entry"] else 1.0)
        * float(item["amount"])
        * float(item["safe_price"])
        for item in [*entries, exit_order]
    ]
    fees = [
        abs(float(item["amount"]) * float(item["safe_price"])) * 0.0005
        for item in [*entries, exit_order]
    ]
    profit = float(f"{math.fsum(cashflows) - math.fsum(fees) + funding_fees:.8f}")
    open_value = entry_cash * (1 - 0.0005 if is_short else 1 + 0.0005)
    ratio = float(f"{(profit / open_value) * 2.0:.8f}")
    stake = entry_cash / 2.0
    return {
        "pair": PAIR,
        "stake_amount": stake,
        "max_stake_amount": stake,
        "amount": total_amount,
        "open_date": datetime.fromtimestamp(open_timestamp / 1000, tz=UTC).isoformat(),
        "close_date": datetime.fromtimestamp(close_timestamp / 1000, tz=UTC).isoformat(),
        "open_rate": 100.0,
        "close_rate": exit_price,
        "fee_open": 0.0005,
        "fee_close": 0.0005,
        "trade_duration": (close_timestamp - open_timestamp) // 60_000,
        "profit_ratio": ratio,
        "profit_abs": profit,
        "exit_reason": exit_reason,
        "initial_stop_loss_abs": 0.0,
        "initial_stop_loss_ratio": -1.0,
        "stop_loss_abs": 0.0,
        "stop_loss_ratio": -1.0,
        "min_rate": 99.0,
        "max_rate": max(101.0, exit_price),
        "is_open": False,
        "enter_tag": f"episode-{ordinal}",
        "leverage": 2.0,
        "is_short": is_short,
        "open_timestamp": open_timestamp,
        "close_timestamp": close_timestamp,
        "orders": [*entries, exit_order],
        "funding_fees": funding_fees,
    }


def _drawdown(trades: list[dict[str, object]]) -> float:
    cumulative = high = worst = relative = 0.0
    for ordinal, trade in sorted(
        enumerate(trades), key=lambda item: (item[1]["close_timestamp"], item[0])
    ):
        del ordinal
        cumulative += float(trade["profit_abs"])
        high = max(high, cumulative)
        drawdown = cumulative - high
        if drawdown < worst:
            worst = drawdown
            relative = -drawdown / (1000.0 + high)
    return relative


def _stats(trades: list[dict[str, object]]) -> dict[str, object]:
    profits = [float(trade["profit_abs"]) for trade in trades]
    total = math.fsum(profits)
    long_total = math.fsum(
        float(trade["profit_abs"]) for trade in trades if not trade["is_short"]
    )
    short_total = math.fsum(
        float(trade["profit_abs"]) for trade in trades if trade["is_short"]
    )
    wins = [value for value in profits if value > 0]
    losses = [value for value in profits if value < 0]
    gross_profit = math.fsum(wins)
    gross_loss = -math.fsum(losses)
    profit_factor = gross_profit / gross_loss if gross_loss else 0.0
    average_win = gross_profit / len(wins) if wins else 0.0
    average_loss = gross_loss / len(losses) if losses else 0.0
    expectancy = (len(wins) / len(profits)) * average_win - (
        len(losses) / len(profits)
    ) * average_loss
    total_row = {
        "key": "TOTAL",
        "trades": len(trades),
        "profit_total_abs": total,
        "profit_total": total / 1000.0,
        "profit_factor": profit_factor,
    }
    stats = {
        "trades": trades,
        "total_trades": len(trades),
        "trade_count_long": sum(not trade["is_short"] for trade in trades),
        "trade_count_short": sum(bool(trade["is_short"]) for trade in trades),
        "starting_balance": 1000.0,
        "stake_amount": 100.0,
        "profit_total_abs": total,
        "profit_total": total / 1000.0,
        "profit_total_long_abs": long_total,
        "profit_total_short_abs": short_total,
        "profit_total_long": long_total / 1000.0,
        "profit_total_short": short_total / 1000.0,
        "expectancy": expectancy,
        "profit_factor": profit_factor,
        "max_drawdown_account": _drawdown(trades),
        "results_per_pair": [total_row],
        "timerange": WINDOWS["development"].timerange,
        "timeframe": "1h",
        "backtest_start_ts": WINDOWS["development"].start_ms,
        "backtest_end_ts": WINDOWS["development"].end_ms,
        **{name: 0 for name in validator.COUNTER_FIELDS},
    }
    return {
        "strategy": {STRATEGY: stats},
        "strategy_comparison": [
            {**total_row, "key": STRATEGY, "max_drawdown_account": stats["max_drawdown_account"]}
        ],
    }

def _write_zip(
    path: Path,
    role: str,
    result: dict[str, object],
    request: dict[str, object],
    replay: tuple[dict, dict, dict[str, bytes]],
    execution: dict[str, object],
) -> tuple[Path, Path, bytes]:
    stem = STEM_B if role == "B" else STEM_C
    source, parameters, strategy_evidence = _strategy_material(role)
    replay_manifest, binding, replay_members = replay
    revision = _revision(
        role,
        request,
        replay_manifest["data_snapshot"],
        binding,
        strategy_evidence,
        execution,
    )
    metadata = {STRATEGY: {"run_id": f"run-{role}", "experiment_revision": revision}}
    archive_config = copy.deepcopy(contract.EXPECTED_EFFECTIVE_CONFIG)
    archive_config["timerange"] = WINDOWS["development"].timerange
    archive_config["backtest_cache"] = "none"
    archive_config["enable_protections"] = False
    archive_config["api_server"]["jwt_secret_key"] = "***"
    archive_config["api_server"]["password"] = "***"
    archive_config["api_server"]["ws_token"] = "***"
    with zipfile.ZipFile(path, "w") as archive:
        archive.writestr(f"{stem}.json", json.dumps(result))
        archive.writestr(f"{stem}_config.json", json.dumps(archive_config))
        archive.writestr(f"{stem}_{STRATEGY}.py", source)
        archive.writestr(f"{stem}_{STRATEGY}_evidence.json", _backend_canon(parameters))
        if role == "B":
            archive.writestr("data_snapshot/manifest.json", _backend_canon(replay_manifest))
            for member, raw in sorted(replay_members.items()):
                archive.writestr(member, raw, compress_type=zipfile.ZIP_STORED)
    metadata_path = path.with_name(f"{stem}.meta.json")
    metadata_path.write_text(json.dumps(metadata), encoding="utf-8")
    return path, metadata_path, source


def _trade_sets(primary_rows: list[OhlcvRow]):
    states = validator.build_indicator_states(primary_rows)
    window = WINDOWS["development"]
    ledger = validator.derive_episode_ledger(states, window.start_ms, window.end_ms)
    episodes = sorted(
        (
            episode
            for episode in ledger.episodes
            if episode.episode_id in set(ledger.scored_armed)
            and episode.first_completion_ms + HOUR < window.end_ms
        ),
        key=lambda item: item.first_completion_ms,
    )
    baseline = [
        _trade(
            index,
            episode.side,
            episode.first_completion_ms + HOUR,
            episode.first_completion_ms + 2 * HOUR,
            -5.0 if index == 0 else 0.5,
        )
        for index, episode in enumerate(episodes[:24])
    ]
    candidate = []
    for index, episode in enumerate(episodes[:24]):
        stop = index < 6
        candidate.append(
            _trade(
                100 + index,
                episode.side,
                episode.first_completion_ms + HOUR,
                episode.first_completion_ms + 2 * HOUR,
                -0.3 if stop else 1.2,
                "stop_loss" if stop else "exit_signal",
            )
        )
    return baseline, candidate

def _rewrite_manifest(path: Path, value: dict[str, object]) -> None:
    path.write_bytes(_canon(value))


def _rehash_record(value: dict[str, object], kind: str, path: Path) -> None:
    raw = path.read_bytes()
    record = next(item for item in value["files"] if item["kind"] == kind)
    record["bytes"] = len(raw)
    record["sha256"] = hashlib.sha256(raw).hexdigest()


def _rewrite_zip(path: Path, mutate) -> None:
    with zipfile.ZipFile(path) as source:
        members = [(item.filename, source.read(item), item.compress_type) for item in source.infolist()]
    members = mutate(members)
    with zipfile.ZipFile(path, "w") as target:
        for name, raw, compression in members:
            target.writestr(name, raw, compress_type=compression)

    short = copy.deepcopy(long_trade)
    short["is_short"] = True
    short["orders"][0]["ft_order_side"] = "sell"
    short["orders"][1]["ft_order_side"] = "sell"

def test_b_and_c_archived_configs_must_be_identical(tmp_path: Path) -> None:
    manifest, value, files = _manifest(tmp_path)
    c_zip = files["c_result_zip"]

    def mutate(members):
        changed = []
        for name, raw, compression in members:
            if name.endswith("_config.json"):
                config = json.loads(raw)
                config["unbound_extra"] = "different"
                raw = json.dumps(config).encode()
            changed.append((name, raw, compression))
        return changed

    _rewrite_zip(c_zip, mutate)
    _rehash_record(value, "c_result_zip", c_zip)
    _rewrite_manifest(manifest, value)
    outcome = validate_manifest(manifest)
    assert outcome.classification == CLASS_INVALID
    assert outcome.reasons == ("cross_config",)

@pytest.mark.parametrize("fault", ["path", "hash", "bytes", "inventory", "order"])
def test_manifest_path_hash_bytes_and_inventory_fail_closed(tmp_path: Path, fault: str) -> None:
    manifest, value, _files = _manifest(tmp_path)
    if fault == "path":
        value["files"][0]["path"] = "relative.json"
    elif fault == "hash":
        value["files"][0]["sha256"] = "0" * 64
    elif fault == "bytes":
        value["files"][0]["bytes"] += 1
    elif fault == "inventory":
        value["files"].pop()

    _rewrite_manifest(manifest, value)
    outcome = validate_manifest(manifest)
    assert outcome.classification == CLASS_INVALID
    assert outcome.reasons[0].startswith("manifest_") or "file_" in outcome.reasons[0]

def test_canonical_report_stdout_and_exit_zero_for_semantic_classes(
    tmp_path: Path, monkeypatch, capsys
) -> None:
    manifest, _value, _files = _manifest(tmp_path)
    for classification in (CLASS_INVALID, CLASS_INSUFFICIENT, CLASS_REJECTED, CLASS_PENDING):
        report = tmp_path / f"{classification}.json"
        monkeypatch.setattr(
            validator,
            "validate_manifest",
            lambda _path, value=classification: validator.ValidationOutcome(
                value,
                ("test",),
                {"runtime_security": _runtime_security()},
            ),
        )
        assert validator._main(["--manifest", str(manifest), "--report", str(report)]) == 0
        assert capsys.readouterr().out == classification + "\n"
        raw = report.read_bytes()
        assert raw.endswith(b"\n") and not raw.endswith(b"\n\n") and b"\r" not in raw
        assert raw == validator._canonical_json(json.loads(raw))
        payload = json.loads(raw)
        assert set(payload) == {
            "schema_version",
            "contract",
            "classification",
            "reasons",
            "audit",
        }
        assert type(payload["schema_version"]) is int and payload["schema_version"] == 1
        assert payload["contract"] == validator.REPORT_CONTRACT
        assert payload["classification"] == classification
        assert payload["reasons"] == ["test"]
        assert payload["audit"] == {"runtime_security": _runtime_security()}
