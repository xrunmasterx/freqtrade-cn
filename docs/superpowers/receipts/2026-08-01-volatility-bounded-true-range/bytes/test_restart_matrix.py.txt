def make_ohlcv(
    rows: int,
    *,
    start: str = "2026-01-01T00:00:00Z",
    volume: float = 10.0,
) -> pd.DataFrame:
    dates = pd.date_range(start, periods=rows, freq="1h", tz="UTC")
    values = np.arange(rows, dtype=float)
    opens = 100.0 + values
    return pd.DataFrame(
        {
            "date": dates,
            "open": opens,
            "high": opens + 2.0,
            "low": opens - 2.0,
            "close": opens + 1.0,
            "volume": np.full(rows, volume, dtype=float),
        }
    )


def route_spec(frame: pd.DataFrame, route: str = "futures") -> dict[str, Any]:
    step = matrix.HOUR_NS if route != "funding" else 8 * matrix.HOUR_NS
    return {
        "route": route,
        "rows": len(frame),
        "start_ns": matrix._timestamp_ns(frame.iloc[0]["date"]),
        "end_ns": matrix._timestamp_ns(frame.iloc[-1]["date"]),
        "step_ns": step,
    }


def output_row(date: pd.Timestamp, source: pd.Timestamp | pd.NaT) -> dict[str, Any]:
    row: dict[str, Any] = {column: math.nan for column in matrix.FLOAT_COLUMNS}
    row.update({column: False for column in matrix.BOOL_COLUMNS})
    row.update({column: 0 for column in matrix.SIGNAL_COLUMNS})
    row["date"] = date
    row["source_bucket_start"] = source
    return row


def make_boundary_frames(offset: int, rows: int = 64) -> tuple[pd.DataFrame, pd.DataFrame]:
    suffix = make_ohlcv(rows, start=f"2026-01-01T{offset:02d}:00:00Z")
    indicator_position = matrix.BOUNDARY_ROWS[offset]["indicator"] - 1
    output = pd.DataFrame(
        [output_row(date, pd.NaT) for date in suffix["date"]], index=suffix.index
    )
    output["source_bucket_start"] = pd.Series(
        pd.NaT, index=output.index, dtype="datetime64[ns, UTC]"
    )
    for position in range(indicator_position, rows):
        completion = indicator_position + 3 * ((position - indicator_position) // 3)
        source = suffix.iloc[completion]["date"] - pd.Timedelta(hours=2)
        output.at[position, "source_bucket_start"] = source
        for column_index, column in enumerate(matrix.FLOAT_COLUMNS, start=1):
            output.at[position, column] = float(column_index)
        output.at[position, "bucket_ok"] = True
        output.at[position, "valid"] = True
    return suffix, output


def make_signature(source_ns: int) -> dict[str, Any]:
    unsigned = {
        "float_nan_mask": "0" * len(matrix.FLOAT_COLUMNS),
        "float_hex": {
            column: float(index + 1).hex()
            for index, column in enumerate(matrix.FLOAT_COLUMNS)
        },
        "booleans": {column: False for column in matrix.BOOL_COLUMNS},
        "signals": {column: 0 for column in matrix.SIGNAL_COLUMNS},
        "source_bucket_start_ns": source_ns,
    }
    return {**unsigned, "sha256": matrix._json_sha256(unsigned)}


def make_source_identity(source_ns: int) -> dict[str, Any]:
    members = [
        {
            "timestamp_ns": source_ns + index * matrix.HOUR_NS,
            "row_sha256": hashlib.sha256(f"member-{index}".encode()).hexdigest(),
        }
        for index in range(3)
    ]
    return {
        "source_bucket_start_ns": source_ns,
        "constituents": members,
        "bucket_rows_sha256": hashlib.sha256(b"three-members").hexdigest(),
    }


def make_boundary_evidence(
    start_offset: int, input_start_ns: int, source_ns: int
) -> dict[str, Any]:
    signature = make_signature(source_ns)
    evidence: dict[str, Any] = {"start_offset": start_offset}
    for kind in ("indicator", "edge"):
        rows = matrix.BOUNDARY_ROWS[start_offset][kind]
        evidence[kind] = {
            "expected_raw_rows": rows,
            "actual_raw_rows": rows,
            "timestamp_ns": input_start_ns + (rows - 1) * matrix.HOUR_NS,
            "earlier_fail_closed": True,
            "signature": signature,
            "reference_signature": signature,
            "compared_fields": list(
                matrix.INDICATOR_BOUNDARY_FIELDS
                if kind == "indicator"
                else matrix.EDGE_BOUNDARY_FIELDS
            ),
            "equal": True,
        }
    return evidence

def write_small_shards(root: Path, contract: matrix.MergeContract) -> None:
    global_identity = {"admission_sha256": "a" * 64, "test": "synthetic"}
    end_rehashes = {"strategy": matrix.STRATEGY_SHA256, "raw": "b" * 64}
    for shard_id in range(contract.shard_count):
        directory = root / f"shard-{shard_id}"
        directory.mkdir()
        records: list[dict[str, Any]] = []
        resolver_ordinal = 0
        for ordinal in range(shard_id, contract.target_count, contract.shard_count):
            target_ns = contract.target_start_ns + ordinal * matrix.HOUR_NS
            target = {
                "ordinal": ordinal,
                "timestamp_ns": target_ns,
                "utc_offset": ordinal % 3,
                "witness": ordinal in contract.witness_ordinals,
            }
            source_lag_hours = {0: 3, 1: 4, 2: 2}[ordinal % 3]
            source_ns = target_ns - source_lag_hours * matrix.HOUR_NS
            signature = make_signature(source_ns)
            source_identity = make_source_identity(source_ns)
            reference_ordinal = resolver_ordinal + 1
            for profile in matrix.PROFILE_ORDER:
                resolver_ordinal += 1
                labels = list(matrix.PROFILE_LABELS[profile])
                if profile == "full":
                    row_count = 1_000 + ordinal
                    requested = {"full-reference": row_count}
                    comparison = None
                    boundaries = None
                else:
                    row_count = int(profile.removeprefix("suffix-"))
                    requested = {label: row_count for label in labels}
                    payload = matrix._comparison_payload(
                        signature, signature, source_identity, source_identity
                    )
                    comparison = {
                        **payload,
                        "reference_resolver_ordinal": reference_ordinal,
                        "equal": True,
                        "sha256": matrix._json_sha256(payload),
                    }
                    input_start_ns = target_ns - (row_count - 1) * matrix.HOUR_NS
                    start_hour = (input_start_ns // matrix.HOUR_NS) % 3
                    boundaries = make_boundary_evidence(
                        start_hour, input_start_ns, source_ns
                    )
                records.append(
                    {
                        "protocol": matrix.PROTOCOL,
                        "shard_id": shard_id,
                        "shard_count": contract.shard_count,
                        "target": target,
                        "profile": profile,
                        "labels": labels,
                        "requested_lengths": requested,
                        "resolver": {
                            "ordinal": resolver_ordinal,
                            "path": str(matrix.EXPECTED_STRATEGY),
                            "source_sha256": matrix.STRATEGY_SHA256,
                            "source_size": matrix.STRATEGY_SIZE,
                            "startup_candle_count": 499,
                            "fresh_instance": True,
                            "fresh_dynamic_class": True,
                        },
                        "input": {
                            "row_count": row_count,
                            "start_ns": target_ns - (row_count - 1) * matrix.HOUR_NS,
                            "end_ns": target_ns,
                            "inventory_sha256": hashlib.sha256(
                                f"{ordinal}/{profile}".encode()
                            ).hexdigest(),
                            "inventory_bytes": row_count * 10,
                        },
                        "signature": signature,
                        "source_identity": source_identity,
                        "comparison": comparison,
                        "boundaries": boundaries,
                        "mismatch_count": 0,
                    }
                )
        records_bytes = b"".join(matrix._canonical_json_bytes(record) for record in records)
        ordinals = list(range(shard_id, contract.target_count, contract.shard_count))
        target_count = len(ordinals)
        done = {
            "protocol": matrix.PROTOCOL,
            "shard_id": shard_id,
            "shard_count": contract.shard_count,
            "target_count": target_count,
            "target_ordinals_sha256": matrix._json_sha256(ordinals),
            "record_count": len(records),
            "resolver_calls": len(records),
            "comparison_count": target_count * 4,
            "mismatch_count": 0,
            "profile_counts": {profile: target_count for profile in matrix.PROFILE_ORDER},
            "label_counts": {
                "499": target_count,
                "600": target_count,
                "799": target_count,
                "cold599": target_count,
                "explicit599": target_count,
                "full-reference": target_count,
                "warm599": target_count,
            },
            "witness_ordinals": [
                value for value in ordinals if value in contract.witness_ordinals
            ],
            "records_sha256": hashlib.sha256(records_bytes).hexdigest(),
            "records_bytes": len(records_bytes),
            "global_identity": global_identity,
            "end_rehashes": end_rehashes,
            "complete": True,
        }
        (directory / "records.jsonl").write_bytes(records_bytes)
        (directory / "done.json").write_bytes(matrix._canonical_json_bytes(done))


def read_small_records(path: Path) -> list[dict[str, Any]]:
    return [json.loads(line) for line in path.read_bytes().splitlines()]


def write_small_records(shard: Path, records: list[dict[str, Any]]) -> None:
    records_bytes = b"".join(matrix._canonical_json_bytes(record) for record in records)
    done, _ = matrix._read_canonical_json(shard / "done.json")
    done["records_sha256"] = hashlib.sha256(records_bytes).hexdigest()
    done["records_bytes"] = len(records_bytes)
    (shard / "records.jsonl").write_bytes(records_bytes)
    (shard / "done.json").write_bytes(matrix._canonical_json_bytes(done))

        (worktree / relative).mkdir(exist_ok=True)
    (worktree / "freqtrade/freqtrade").mkdir(exist_ok=True)

def test_required_env_path_rejects_missing_relative_and_substituted(tmp_path: Path) -> None:
    expected = tmp_path / "expected.txt"
    expected.write_text("identity", encoding="utf-8")
    with pytest.raises(matrix.ContractError, match="required"):
        matrix._validate_exact_path(
            None, environment_name="TEST_PATH", expected=expected, kind="file"
        )
    with pytest.raises(matrix.ContractError, match="absolute"):
        matrix._validate_exact_path(
            "expected.txt", environment_name="TEST_PATH", expected=expected, kind="file"
        )
    substitute = tmp_path / "substitute.txt"
    substitute.write_text("identity", encoding="utf-8")
    with pytest.raises(matrix.ContractError, match="substituted"):
        matrix._validate_exact_path(
            substitute, environment_name="TEST_PATH", expected=expected, kind="file"
        )
    assert (
        matrix._validate_exact_path(
            expected, environment_name="TEST_PATH", expected=expected, kind="file"
        )
        == expected
    )

def test_repository_escape_blocks_git_and_freqtrade_import(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    make_repository_tree(tmp_path)
    outside = tmp_path.parent / f"{tmp_path.name}-outside-backend"
    outside.mkdir()
    real_directory = matrix._directory_no_links
    git_calls: list[Path] = []
    import_calls: list[str] = []

    def escaped_backend(path: Path, label: str) -> Path:
        resolved = real_directory(path, label)
        return outside.resolve(strict=True) if label == "backend repository" else resolved

    def unreachable_git(repository: Path, *arguments: str) -> str:
        git_calls.append(repository)
        raise AssertionError(f"Git reached {repository}: {arguments}")

    def unreachable_import(name: str) -> Any:
        import_calls.append(name)
        raise AssertionError(f"import reached {name}")

    monkeypatch.setattr(matrix, "_directory_no_links", escaped_backend)
    monkeypatch.setattr(matrix, "_run_git", unreachable_git)
    monkeypatch.setattr(matrix.importlib, "import_module", unreachable_import)
    with pytest.raises(matrix.ContractError, match="escapes the validated worktree"):
        matrix._validate_repository_checkout(tmp_path)
    with pytest.raises(matrix.ContractError, match="escapes the validated worktree"):
        matrix._import_freqtrade_core(tmp_path)
    assert git_calls == []
    assert import_calls == []

def test_timestamp_matching_explicit_pyc_is_rejected_before_git_or_import(

    source.write_text("VALUE = 1\n", encoding="utf-8")
    cached = Path(
        py_compile.compile(
            str(source),
            doraise=True,
            invalidation_mode=py_compile.PycInvalidationMode.TIMESTAMP,

    import_calls: list[str] = []

    monkeypatch.setattr(matrix.importlib, "import_module", unreachable_import)
    with pytest.raises(matrix.ContractError, match=r"__pycache__|cached bytecode"):

    assert import_calls == []

def test_run_shard_command_writes_safe_failure_once_and_cannot_resume(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
    failure: Exception,
    expected_exit: int,
    expected_kind: str,
) -> None:
    state = matrix.AdmissionState(
        guard=matrix._ADMISSION_GUARD,
        summary={},
        primary_frame=None,
        paths={},
        raw_hashes={},
        trading_mode=None,
        margin_mode=None,
    )
    admit_calls = 0

    def fake_admit() -> matrix.AdmissionState:
        nonlocal admit_calls
        admit_calls += 1
        return state

    def fail_run_shard(
        _state: matrix.AdmissionState,
        shard_id: int,
        _output: Path,
        *,
        failure_context: dict[str, Any],
    ) -> dict[str, Any]:

                "next_resolver_ordinal": 8,

        )

        matrix,
        "_admission_identity",

    )

    }
    assert set(evidence["context"]["end_identities"]) == {"expected", "actual"}
    assert b"SECRET_PAYLOAD_MUST_NOT_APPEAR" not in original

    assert matrix.main(command) == 2
    assert (output / "failure.json").read_bytes() == original
    assert {item.name for item in output.iterdir()} == {"failure.json"}
    assert admit_calls == 1

@pytest.mark.parametrize(
    ("mutation", "message"),
    [
        ("tamper", "records.jsonl hash differs"),
        ("truncate", "records are truncated"),
        ("extra", "records contain extra data"),
        ("order", "record order differs"),
    ],
)
def test_streaming_merge_rejects_per_shard_tamper_truncation_extra_and_order(
    tmp_path: Path,
    small_contract: matrix.MergeContract,
    mutation: str,
    message: str,
) -> None:
    shards = tmp_path / "shards"
    shards.mkdir()
    write_small_shards(shards, small_contract)
    shard = shards / "shard-3"
    records_path = shard / "records.jsonl"
    lines = records_path.read_bytes().splitlines(keepends=True)
    if mutation == "tamper":
        record = json.loads(lines[0])
        record["input"]["inventory_sha256"] = "c" * 64
        lines[0] = matrix._canonical_json_bytes(record)
        records_path.write_bytes(b"".join(lines))
    elif mutation == "truncate":
        records_path.write_bytes(b"".join(lines[:-1]))
    elif mutation == "extra":
        records_path.write_bytes(b"".join([*lines, lines[-1]]))
    elif mutation == "order":
        lines[0], lines[1] = lines[1], lines[0]
        records_path.write_bytes(b"".join(lines))
        done, _ = matrix._read_canonical_json(shard / "done.json")
        done["records_sha256"] = hashlib.sha256(records_path.read_bytes()).hexdigest()
        (shard / "done.json").write_bytes(matrix._canonical_json_bytes(done))
    with pytest.raises(matrix.ContractError, match=message):
        matrix.verify_merge(
            shards, tmp_path / "summary.json", contract=small_contract
        )

def test_streaming_merge_digest_is_global_target_profile_order(
    tmp_path: Path, small_contract: matrix.MergeContract
) -> None:
    shards = tmp_path / "shards"
    shards.mkdir()
    write_small_shards(shards, small_contract)
    shard_lines = {
        shard_id: (shards / f"shard-{shard_id}/records.jsonl")
        .read_bytes()
        .splitlines(keepends=True)
        for shard_id in range(small_contract.shard_count)
    }
    positions = {shard_id: 0 for shard_id in range(small_contract.shard_count)}
    expected_digest = hashlib.sha256()
    expected_bytes = 0
    for ordinal in range(small_contract.target_count):
        shard_id = ordinal % small_contract.shard_count
        start = positions[shard_id]
        for line in shard_lines[shard_id][start : start + len(matrix.PROFILE_ORDER)]:
            expected_digest.update(line)
            expected_bytes += len(line)
        positions[shard_id] += len(matrix.PROFILE_ORDER)

    summary = matrix.verify_merge(
        shards, tmp_path / "summary.json", contract=small_contract
    )
    assert summary["merged_records_sha256"] == expected_digest.hexdigest()
    assert summary["merged_records_bytes"] == expected_bytes

    with pytest.raises(matrix.ContractError, match="symbolic/reparse"):
        matrix.verify_merge(
            shards, tmp_path / "summary.json", contract=small_contract
        )

def test_done_binds_shared_identities_without_repeating_them_in_records(
    tmp_path: Path, small_contract: matrix.MergeContract
) -> None:
    shards = tmp_path / "shards"
    shards.mkdir()
    write_small_shards(shards, small_contract)
    shard = shards / "shard-3"
    records = read_small_records(shard / "records.jsonl")
    assert "global_identity" not in records[0]
    assert "end_rehashes" not in records[0]
    done, _ = matrix._read_canonical_json(shard / "done.json")
    assert "global_identity" in done
    assert "end_rehashes" in done
    done["global_identity"] = {"admission_sha256": "c" * 64, "test": "mixed"}
    (shard / "done.json").write_bytes(matrix._canonical_json_bytes(done))
    with pytest.raises(matrix.ContractError, match="shard global identities differ"):
        matrix.verify_merge(shards, tmp_path / "summary.json", contract=small_contract)


def test_merge_rejects_tampered_records(

def test_merge_recomputes_comparison_instead_of_trusting_equal_flag(
    tmp_path: Path, small_contract: matrix.MergeContract
) -> None:
    shards = tmp_path / "shards"
    shards.mkdir()
    write_small_shards(shards, small_contract)
    shard = shards / "shard-0"
    records = read_small_records(shard / "records.jsonl")
    records[1]["signature"]["float_hex"][matrix.FLOAT_COLUMNS[0]] = 99.0.hex()
    unsigned = dict(records[1]["signature"])
    unsigned.pop("sha256")
    records[1]["signature"]["sha256"] = matrix._json_sha256(unsigned)
    write_small_records(shard, records)
    with pytest.raises(matrix.ContractError, match="merged target signature differs"):
        matrix.verify_merge(shards, tmp_path / "summary.json", contract=small_contract)
